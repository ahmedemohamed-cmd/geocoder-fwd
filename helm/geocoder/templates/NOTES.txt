{{- if .Values.geocoder.ingress.enabled }}
{{- range .Values.geocoder.ingress.hosts }}
  http://{{ .host }}{{ (index .paths 0).path }}
{{- end }}
{{- else if contains "LoadBalancer" .Values.geocoder.service.type }}
  kubectl get svc {{ include "geocoder.fullname" . }}-geocoder
{{- else }}
  kubectl port-forward svc/{{ include "geocoder.fullname" . }}-geocoder 8000:{{ .Values.geocoder.service.port }}

{{- end }}

Trigger the downloader immediately (don't wait for the monthly cron):
  kubectl create job --from=cronjob/{{ include "geocoder.fullname" . }}-downloader manual-download-$(date +%s)

Trigger the watcher immediately:
  kubectl create job --from=cronjob/{{ include "geocoder.fullname" . }}-watcher manual-parse-$(date +%s)

View pipeline logs:
  kubectl logs -l app.kubernetes.io/instance={{ .Release.Name }} --tail=50 -f
